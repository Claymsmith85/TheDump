#Requires -Modules Az.Accounts, Az.Storage, Az.Network, Az.PrivateDns
<#
.SYNOPSIS
    Adds a 'file' private endpoint to a storage account, reusing the subnet of an
    existing private endpoint, and registers it in privatelink.file.core.windows.net.

.EXAMPLE
    .\New-StorageFilePrivateEndpoint.ps1 `
        -SubscriptionId "<sub-id>" `
        -StorageResourceGroup "<sa-rg>" `
        -StorageAccountName "sacoredataspksfpstorage" `
        -ReferenceEndpointName "pep-core-data-sacoredataspksfpstorage-sftpstorage-infra000" `
        -DnsZoneResourceGroup "<dns-rg>" `
        -DnsZoneSubscriptionId "<hub-sub-id>"
#>
[CmdletBinding(SupportsShouldProcess)]
param(
    [Parameter(Mandatory)] [string] $SubscriptionId,
    [Parameter(Mandatory)] [string] $StorageResourceGroup,
    [Parameter(Mandatory)] [string] $StorageAccountName,
    [Parameter(Mandatory)] [string] $ReferenceEndpointName,   # existing blob PE; subnet/location/RG are copied from it
    [string] $NewEndpointName,                                 # default: pep-<account>-file
    [string] $DnsZoneResourceGroup,                            # omit to skip DNS registration
    [string] $DnsZoneSubscriptionId,                           # default: same as SubscriptionId
    [string] $TenantId
)

$ErrorActionPreference = 'Stop'
$zoneName = 'privatelink.file.core.windows.net'
if (-not $NewEndpointName)      { $NewEndpointName = "pep-$StorageAccountName-file" }
if (-not $DnsZoneSubscriptionId) { $DnsZoneSubscriptionId = $SubscriptionId }

# --- Context ---------------------------------------------------------------
$ctxArgs = @{ Subscription = $SubscriptionId }
if ($TenantId) { $ctxArgs.Tenant = $TenantId }
Set-AzContext @ctxArgs | Out-Null
Write-Host "Context: $((Get-AzContext).Subscription.Name)"

# --- Storage account -------------------------------------------------------
$sa = Get-AzStorageAccount -ResourceGroupName $StorageResourceGroup -Name $StorageAccountName
Write-Host "Storage account: $($sa.Id)"

# --- Reference endpoint (subnet, location, RG) -----------------------------
$ref = Get-AzPrivateEndpoint | Where-Object Name -eq $ReferenceEndpointName
if (-not $ref) { throw "Reference private endpoint '$ReferenceEndpointName' not found in this subscription." }

$subnet   = Get-AzVirtualNetworkSubnetConfig -ResourceId $ref.Subnet.Id
$location = $ref.Location
$peRg     = $ref.ResourceGroupName
Write-Host "Subnet:   $($ref.Subnet.Id)"
Write-Host "Location: $location   PE resource group: $peRg"

# --- Existing file endpoint check ------------------------------------------
$existing = Get-AzPrivateEndpoint | Where-Object {
    $_.PrivateLinkServiceConnections.PrivateLinkServiceId -contains $sa.Id -and
    $_.PrivateLinkServiceConnections.GroupIds -contains 'file'
}
if ($existing) {
    Write-Warning "A 'file' private endpoint already exists: $($existing.Name). Using it."
    $pe = $existing | Select-Object -First 1
}
else {
    # --- Create endpoint ---------------------------------------------------
    $conn = New-AzPrivateLinkServiceConnection -Name "plsc-$StorageAccountName-file" `
        -PrivateLinkServiceId $sa.Id -GroupId 'file'

    if ($PSCmdlet.ShouldProcess($NewEndpointName, 'Create private endpoint (file)')) {
        $pe = New-AzPrivateEndpoint -Name $NewEndpointName -ResourceGroupName $peRg `
            -Location $location -Subnet $subnet -PrivateLinkServiceConnection $conn
        Write-Host "Created private endpoint: $($pe.Name)"
    }
}

if ($pe) {
    $state = $pe.PrivateLinkServiceConnections[0].PrivateLinkServiceConnectionState.Status
    $ips   = ($pe.CustomDnsConfigs.IpAddresses) -join ', '
    Write-Host "Connection state: $state"
    Write-Host "Private IP(s):    $ips"
    if ($state -ne 'Approved') {
        Write-Warning "Connection is not approved. Approve it under Storage account > Networking > Private endpoint connections."
    }
}

# --- DNS registration ------------------------------------------------------
if ($DnsZoneResourceGroup -and $pe) {
    if ($DnsZoneSubscriptionId -ne $SubscriptionId) { Set-AzContext -Subscription $DnsZoneSubscriptionId | Out-Null }
    $zone = Get-AzPrivateDnsZone -ResourceGroupName $DnsZoneResourceGroup -Name $zoneName -ErrorAction SilentlyContinue
    if ($DnsZoneSubscriptionId -ne $SubscriptionId) { Set-AzContext @ctxArgs | Out-Null }

    if (-not $zone) {
        Write-Warning "Zone '$zoneName' not found in $DnsZoneResourceGroup. Create and link it to the DNS VNet, then rerun."
    }
    else {
        $existingGroup = Get-AzPrivateDnsZoneGroup -ResourceGroupName $peRg -PrivateEndpointName $pe.Name -ErrorAction SilentlyContinue
        if ($existingGroup) {
            Write-Host "DNS zone group already present: $($existingGroup.Name)"
        }
        elseif ($PSCmdlet.ShouldProcess($pe.Name, "Link to $zoneName")) {
            $cfg = New-AzPrivateDnsZoneConfig -Name 'file' -PrivateDnsZoneId $zone.ResourceId
            New-AzPrivateDnsZoneGroup -ResourceGroupName $peRg -PrivateEndpointName $pe.Name `
                -Name 'default' -PrivateDnsZoneConfig $cfg | Out-Null
            Write-Host "Registered in $zoneName"
        }
    }
}
else {
    Write-Host "DNS registration skipped."
}

# --- Verify from this machine ----------------------------------------------
$fqdn = "$StorageAccountName.file.core.windows.net"
Write-Host "`nVerification for $fqdn"
Resolve-DnsName $fqdn -ErrorAction SilentlyContinue |
    Where-Object { $_.IPAddress } | Select-Object Name, IPAddress | Format-Table -AutoSize
Test-NetConnection $fqdn -Port 445 -InformationLevel Quiet |
    ForEach-Object { Write-Host "Port 445 reachable: $_" }
